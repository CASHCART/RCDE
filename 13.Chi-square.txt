#test of independence

data <- matrix(c(30, 10, 25, 15),
               nrow = 2, byrow = TRUE)

colnames(data) <- c("Like", "Dislike")
rownames(data) <- c("Men", "Women")

chisq_test <- chisq.test(data)

print(chisq_test)


#goodness of fit test

observed <- c(10, 8, 12, 14, 6, 10)

expected <- rep(10, 6)

chisq_gof <- chisq.test(
  observed,
  p = expected / sum(expected)
)

print(chisq_gof)
