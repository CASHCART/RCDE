#include <stdio.h>

int main() {
    int a, b, n;
    int i, sum = 0;

    printf("Enter values of a, b and n: ");

    if (scanf("%d%d%d", &a, &b, &n) != 3) {
        return 1;
    }

    /* 1. Constant Folding */
    int constant_result = 10 * 20;

    printf("\n1. Constant Folding:");
    printf("\nResult = %d\n", constant_result);

    /* 2. Constant Propagation */
    int x = 10;
    int y = x + 5;

    printf("\n2. Constant Propagation:");
    printf("\nValue of y = %d\n", y);

    /* 3. Algebraic Simplification */
    int z = a * 1 + 0;

    printf("\n3. Algebraic Simplification:");
    printf("\nOptimized value of z = %d\n", z);

    /* 4. Common Subexpression Elimination */
    int p = a + b;
    int result1 = p * 2;
    int result2 = p * 3;

    printf("\n4. Common Subexpression Elimination:");
    printf("\nResult1 = %d", result1);
    printf("\nResult2 = %d\n", result2);

    /* 5. Dead Code Elimination */
    int useful = a + b;
    int unused = a * b;

    printf("\n5. Dead Code Elimination:");
    printf("\nUseful result = %d\n", useful);

    /* 6. Loop Unrolling */
    sum = 0;

    for (i = 0; i + 3 < n; i += 4) {
        sum += i;
        sum += i + 1;
        sum += i + 2;
        sum += i + 3;
    }

    while (i < n) {
        sum += i;
        i++;
    }

    printf("\n6. Loop Unrolling:");
    printf("\nSum = %d\n", sum);

    /* 7. Loop Fusion / Loop Jamming */
    int sum1 = 0, sum2 = 0;

    for (i = 0; i < n; i++) {
        sum1 += i;
        sum2 += i * i;
    }

    printf("\n7. Loop Fusion/Loop Jamming:");
    printf("\nSum of numbers = %d", sum1);
    printf("\nSum of squares = %d\n", sum2);

    return 0;
}
/*
1. Constant Folding:
Result = 200

2. Constant Propagation:
Value of y = 15

3. Algebraic Simplification:
Optimized value of z = 5

4. Common Subexpression Elimination:
Result1 = 16
Result2 = 24

5. Dead Code Elimination:
Useful result = 8

6. Loop Unrolling:
Sum = 10

7. Loop Fusion/Loop Jamming:
Sum of numbers = 10
Sum of squares = 30
*/
