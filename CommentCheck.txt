#include <stdio.h>
#include <string.h>

int main() {
    char str[200];

    printf("Enter a string: ");
    fgets(str, sizeof(str), stdin);

    int len = strlen(str);

    if (str[len - 1] == '\n') {
        str[len - 1] = '\0';
        len--;
    }

    if (len >= 2 && str[0] == '/' && str[1] == '/') {
        printf("single line comment");
    }
    else if (len >= 4 && str[0] == '/' && str[1] == '*' &&
             str[len - 1] == '/' && str[len - 2] == '*') {
        printf("multi line comment");
    }
    else {
        printf("not a comment");
    }

    return 0;
}
