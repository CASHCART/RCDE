
#include <stdio.h>
#include <string.h>

int main() {
    char exp[100], token[100][20];
    int i, j, n = 0, temp = 1, pass;

    printf("Enter expression: ");
    scanf("%99s", exp);

    // Split expression into tokens
    for(i = 0; exp[i] != '\0'; i++) {
        token[n][0] = exp[i];
        token[n][1] = '\0';
        n++;
    }

    // First handle * and /
    // Then handle + and -
    for(pass = 0; pass < 2; pass++) {
        i = 1;

        while(i < n - 1) {
            if((pass == 0 &&
               (token[i][0] == '*' || token[i][0] == '/')) ||
               (pass == 1 &&
               (token[i][0] == '+' || token[i][0] == '-'))) {

                printf("t%d = %s %s %s\n",
                       temp, token[i-1], token[i], token[i+1]);

                sprintf(token[i-1], "t%d", temp);
                temp++;

                // Remove used operator and operand
                for(j = i; j < n-2; j++) {
                    strcpy(token[j], token[j+2]);
                }

                n -= 2;
                i = 1;
            }
            else {
                i += 2;
            }
        }
    }

    return 0;
}
/*
1. Input: a+b*c
Enter expression: a+b*c
t1 = b * c
t2 = a + t1


2. Input: a*b+c*d
Enter expression: a*b+c*d
t1 = a * b
t2 = c * d
t3 = t1 + t2
*/
